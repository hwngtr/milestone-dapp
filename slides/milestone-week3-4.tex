\begin{frame}[fragile]{2. Public functions in ICampaign}
\footnotesize
\begin{verbatim}
// ICampaign.sol:4-8
/// @dev Deliberately contains no function that lets an administrator move donor funds.

\end{verbatim}
\vspace{0.2em}
\begin{verbatim}
// ICampaign.sol:60,66,75,80,86,91,97
function contribute() external payable;
function deliver(uint8 milestoneId, bytes32 evidenceHash) external;
function object(uint8 milestoneId) external;
function release(uint8 milestoneId) external;
function claimRefund() external;
function withdraw() external;
function finalizeOutcome() external;
\end{verbatim}
\vspace{0.3em}
\small The interface lists seven functions. There is no \texttt{owner}, \texttt{pause}, \texttt{upgrade}, or \texttt{sweep}.
The two payout functions send ETH to a donor or to the creator.
\end{frame}

\begin{frame}[fragile]{3. contribute() and deliver()}

// Campaign.sol:88-96
function contribute() external payable {
    ICampaign.Status current = _sync();
    if (current != ICampaign.Status.Funding) revert WrongStatus(current);
    if (totalRaised + msg.value > goal) revert GoalAlreadyReached();
    if (msg.value == 0) revert ZeroAmount();
    totalRaised += msg.value;
    contributionOf[msg.sender] += msg.value;
    emit Contributed(msg.sender, msg.value, totalRaised);
}
\end{verbatim}

\begin{verbatim}

% ---------------------------------------------------------------- 5. Payouts
\begin{frame}[fragile]{5. claimRefund() and withdraw()}
\scriptsize
\begin{verbatim}
// Campaign.sol:172-185, claimRefund()
ICampaign.Status current = _sync();
if (current == ICampaign.Status.Funding) revert DeadlineNotReached();
if (current == ICampaign.Status.Active || current == ICampaign.Status.Completed)
    revert GoalAlreadyReached();
uint256 amount = contributionOf[msg.sender];
if (amount == 0) revert NothingToWithdraw();
contributionOf[msg.sender] = 0;              // Effects first
refundsWithdrawn += amount;
(bool ok,) = msg.sender.call{value: amount}("");  // Interaction last
if (!ok) revert TransferFailed();
\end{verbatim}
\vspace{0.25em}
\begin{verbatim}
// Campaign.sol:190-197, withdraw() -- onlyCreator
uint256 amount = creatorCredit - creatorWithdrawn;
if (amount == 0) revert NothingToWithdraw();
creatorWithdrawn += amount;
(bool ok,) = msg.sender.call{value: amount}("");
if (!ok) revert TransferFailed();
\end{verbatim}
\vspace{0.3em}
\small Both functions update state before the external call. A second call reverts. No function loops over donors.
\end{frame}

if (deadline <= block.timestamp) revert InvalidDeadline();

    revert LengthMismatch();

\begin{frame}{7. Demo and test results}
\begin{columns}[T,onlytextwidth]
  \begin{column}{0.48\textwidth}
    \small\textbf{\texttt{make demo} on Anvil}
    \begin{enumerate}
      \item Deploy factory, create campaign (10 ETH, milestones 4/3/3)
      \item Donor contributes the goal
      \item Deadline passes, \texttt{finalizeOutcome()}
      \item Creator calls \texttt{deliver(0, hash)}
      \item Window closes, \textbf{donor} calls \texttt{release(0)}
      \item Creator calls \texttt{withdraw()}
      \item \texttt{grep} the source: no admin function
    \end{enumerate}
  \end{column}
  \begin{column}{0.48\textwidth}
    \small\textbf{\texttt{make test}, green}
    \begin{itemize}
      \item 38 unit tests
      \item 4 fuzz invariants over a 5-actor ghost ledger
      \item Exact-half veto boundary tested: 5 of 10 ETH cannot cancel
      \item Veto cancels, refunds pay exact contributions
      \item Second refund and second release revert
    \end{itemize}
  \end{column}
\end{columns}

\vspace{0.4em}
\small In the demo, a donor calls \texttt{release(0)}. The creator does not. No admin role exists.
